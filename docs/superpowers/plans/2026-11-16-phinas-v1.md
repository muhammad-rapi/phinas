# Phinas v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Spotify-style desktop music player (Mac and Windows) with search, playback, personal library, partner-playlist sync, and a Dynamic cover-art theme.

**Architecture:** Electron main process runs three units (Music source via `youtubei.js` plus a local stream proxy, Local library on SQLite, Sync to Supabase). The React renderer reaches them only through a typed IPC API (`window.phinas`). Library changes are written to SQLite first and queued in an `outbox` table; a sync engine flushes the queue to Supabase.

**Tech Stack:** Electron 45, electron-vite 6, Vite 7, React 19, TypeScript, Vitest 5, `node:sqlite` (built into Node 24, which Electron 45 bundles), `youtubei.js` 18, `@supabase/supabase-js` 2.

**Spec:** `docs/superpowers/specs/2026-11-16-phinas-design.md`

**Conventions:** Run all commands from `~/Dev/phinas-desktop`. Tests live in `tests/` and import from `src/`. Commit after every task. The user is not a programmer: UI tasks end with a concrete "what you should see" check.

```ts
import { DatabaseSync } from 'node:sqlite'

const SCHEMA = `
CREATE TABLE IF NOT EXISTS playlists(
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  updated_at INTEGER NOT NULL
);
CREATE TABLE IF NOT EXISTS playlist_tracks(
  playlist_id TEXT NOT NULL REFERENCES playlists(id) ON DELETE CASCADE,
  position INTEGER NOT NULL,
  video_id TEXT NOT NULL,
  title TEXT NOT NULL,
  artist TEXT NOT NULL,
  cover_url TEXT NOT NULL,
  duration_sec INTEGER NOT NULL,
  PRIMARY KEY (playlist_id, position)
);
CREATE TABLE IF NOT EXISTS liked_tracks(
  video_id TEXT PRIMARY KEY,
  title TEXT NOT NULL,
  artist TEXT NOT NULL,
  cover_url TEXT NOT NULL,
  duration_sec INTEGER NOT NULL,
  liked_at INTEGER NOT NULL
);
CREATE TABLE IF NOT EXISTS outbox(
  key TEXT PRIMARY KEY,
  created_at INTEGER NOT NULL
);
`

export function openDb(path: string): DatabaseSync {
  const db = new DatabaseSync(path)
  db.exec('PRAGMA foreign_keys = ON;')
  if (path !== ':memory:') db.exec('PRAGMA journal_mode = WAL;')
  db.exec(SCHEMA)
  return db
}
```

- [ ] **Step 4: Write `src/main/library/library.ts`**

```ts
import { randomUUID } from 'node:crypto'
import type { DatabaseSync } from 'node:sqlite'
import type { Playlist, PlaylistWithTracks, Track } from '../../shared/types'

interface TrackRow {
  video_id: string
  title: string
  artist: string
  cover_url: string
  duration_sec: number
}

const toTrack = (r: TrackRow): Track => ({
  videoId: r.video_id,
  title: r.title,
  artist: r.artist,
  coverUrl: r.cover_url,
  durationSec: r.duration_sec
})

export class Library {
  constructor(
    private db: DatabaseSync,
    private now: () => number = Date.now,
    private newId: () => string = randomUUID
  ) {}

  private tx<T>(fn: () => T): T {
    this.db.exec('BEGIN')
    try {
      const out = fn()
      this.db.exec('COMMIT')
      return out
    } catch (e) {
      this.db.exec('ROLLBACK')
      throw e
    }
  }

  private enqueue(key: string): void {
    this.db
      .prepare('INSERT OR REPLACE INTO outbox(key, created_at) VALUES (?, ?)')
      .run(key, this.now())
  }

  private touch(id: string): void {
    this.db.prepare('UPDATE playlists SET updated_at = ? WHERE id = ?').run(this.now(), id)
    this.enqueue(`playlist:${id}`)
  }

  createPlaylist(name: string): Playlist {
    const p = { id: this.newId(), name, updatedAt: this.now() }
    this.tx(() => {
      this.db
        .prepare('INSERT INTO playlists(id, name, updated_at) VALUES (?, ?, ?)')
        .run(p.id, p.name, p.updatedAt)
      this.enqueue(`playlist:${p.id}`)
    })
    return p
  }

  renamePlaylist(id: string, name: string): void {
    this.tx(() => {
      this.db.prepare('UPDATE playlists SET name = ? WHERE id = ?').run(name, id)
      this.touch(id)
    })
  }

  deletePlaylist(id: string): void {
    this.tx(() => {
      this.db.prepare('DELETE FROM playlists WHERE id = ?').run(id)
      this.enqueue(`playlist:${id}`)
    })
  }

  listPlaylists(): Playlist[] {
    const rows = this.db
      .prepare('SELECT id, name, updated_at FROM playlists ORDER BY updated_at DESC, rowid DESC')
      .all() as { id: string; name: string; updated_at: number }[]
    return rows.map((r) => ({ id: r.id, name: r.name, updatedAt: r.updated_at }))
  }

  getPlaylist(id: string): PlaylistWithTracks | null {
    const p = this.db.prepare('SELECT id, name, updated_at FROM playlists WHERE id = ?').get(id) as
      | { id: string; name: string; updated_at: number }
      | undefined
    if (!p) return null
    const rows = this.db
      .prepare(
        `SELECT video_id, title, artist, cover_url, duration_sec
         FROM playlist_tracks WHERE playlist_id = ? ORDER BY position`
      )
      .all(id) as unknown as TrackRow[]
    return { id: p.id, name: p.name, updatedAt: p.updated_at, tracks: rows.map(toTrack) }
  }

  addTrack(playlistId: string, track: Track): boolean {
    return this.tx(() => {
      const dup = this.db
        .prepare('SELECT 1 FROM playlist_tracks WHERE playlist_id = ? AND video_id = ?')
        .get(playlistId, track.videoId)
      if (dup) return false
      const max = this.db
        .prepare('SELECT COALESCE(MAX(position), -1) AS m FROM playlist_tracks WHERE playlist_id = ?')
        .get(playlistId) as { m: number }
      this.db
        .prepare(
          `INSERT INTO playlist_tracks(playlist_id, position, video_id, title, artist, cover_url, duration_sec)
           VALUES (?, ?, ?, ?, ?, ?, ?)`
        )
        .run(playlistId, max.m + 1, track.videoId, track.title, track.artist, track.coverUrl, track.durationSec)
      this.touch(playlistId)
      return true
    })
  }

  removeTrack(playlistId: string, videoId: string): void {
    this.tx(() => {
      const current = this.getPlaylist(playlistId)
      if (!current) return
      const kept = current.tracks.filter((x) => x.videoId !== videoId)
      this.db.prepare('DELETE FROM playlist_tracks WHERE playlist_id = ?').run(playlistId)
      const insert = this.db.prepare(
        `INSERT INTO playlist_tracks(playlist_id, position, video_id, title, artist, cover_url, duration_sec)
         VALUES (?, ?, ?, ?, ?, ?, ?)`
      )
      kept.forEach((x, i) =>
        insert.run(playlistId, i, x.videoId, x.title, x.artist, x.coverUrl, x.durationSec)
      )
      this.touch(playlistId)
    })
  }

  listLikes(): Track[] {
    const rows = this.db
      .prepare(
        `SELECT video_id, title, artist, cover_url, duration_sec
         FROM liked_tracks ORDER BY liked_at DESC, rowid DESC`
      )
      .all() as unknown as TrackRow[]
    return rows.map(toTrack)
  }

  setLiked(track: Track, liked: boolean): void {
    this.tx(() => {
      if (liked) {
        this.db
          .prepare(
            `INSERT OR REPLACE INTO liked_tracks(video_id, title, artist, cover_url, duration_sec, liked_at)
             VALUES (?, ?, ?, ?, ?, ?)`
          )
          .run(track.videoId, track.title, track.artist, track.coverUrl, track.durationSec, this.now())
      } else {
        this.db.prepare('DELETE FROM liked_tracks WHERE video_id = ?').run(track.videoId)
      }
      this.enqueue('likes')
    })
  }

  pendingKeys(): string[] {
    const rows = this.db.prepare('SELECT key FROM outbox ORDER BY created_at, rowid').all() as {
      key: string
    }[]
    return rows.map((r) => r.key)
  }

  clearKey(key: string): void {
    this.db.prepare('DELETE FROM outbox WHERE key = ?').run(key)
  }
}
```

```ts
export interface Rgb {
  r: number
  g: number
  b: number
}

export const DEFAULT_ACCENT: Rgb = { r: 139, g: 92, b: 246 }

export function rgbToHsl({ r, g, b }: Rgb): { h: number; s: number; l: number } {
  const rn = r / 255
  const gn = g / 255
  const bn = b / 255
  const max = Math.max(rn, gn, bn)
  const min = Math.min(rn, gn, bn)
  const l = (max + min) / 2
  const d = max - min
  if (d === 0) return { h: 0, s: 0, l }
  const s = l > 0.5 ? d / (2 - max - min) : d / (max + min)
  let h: number
  if (max === rn) h = (gn - bn) / d + (gn < bn ? 6 : 0)
  else if (max === gn) h = (bn - rn) / d + 2
  else h = (rn - gn) / d + 4
  return { h: h / 6, s, l }
}

export function hslToRgb(h: number, s: number, l: number): Rgb {
  if (s === 0) {
    const v = Math.round(l * 255)
    return { r: v, g: v, b: v }
  }
  const q = l < 0.5 ? l * (1 + s) : l + s - l * s
  const p = 2 * l - q
  const f = (t: number): number => {
    let x = t
    if (x < 0) x += 1
    if (x > 1) x -= 1
    if (x < 1 / 6) return p + (q - p) * 6 * x
    if (x < 1 / 2) return q
    if (x < 2 / 3) return p + (q - p) * (2 / 3 - x) * 6
    return p
  }
  return {
    r: Math.round(f(h + 1 / 3) * 255),
    g: Math.round(f(h) * 255),
    b: Math.round(f(h - 1 / 3) * 255)
  }
}

/** RGBA flat array (like ImageData.data). Prefers vivid colors over gray areas. */
export function dominantColor(data: ArrayLike<number>): Rgb {
  const buckets = new Map<number, { n: number; r: number; g: number; b: number; score: number }>()
  for (let i = 0; i + 3 < data.length; i += 4) {
    if (data[i + 3] < 128) continue
    const r = data[i]
    const g = data[i + 1]
    const b = data[i + 2]
    const key = ((r >> 4) << 8) | ((g >> 4) << 4) | (b >> 4)
    const { s } = rgbToHsl({ r, g, b })
    const e = buckets.get(key) ?? { n: 0, r: 0, g: 0, b: 0, score: 0 }
    e.n++
    e.r += r
    e.g += g
    e.b += b
    e.score += 1 + s * 4
    buckets.set(key, e)
  }
  let best: { n: number; r: number; g: number; b: number; score: number } | null = null
  for (const e of buckets.values()) if (!best || e.score > best.score) best = e
  if (!best) return DEFAULT_ACCENT
  return { r: Math.round(best.r / best.n), g: Math.round(best.g / best.n), b: Math.round(best.b / best.n) }
}

const clamp = (x: number, lo: number, hi: number): number => Math.min(hi, Math.max(lo, x))

/** Keep the accent readable on dark panels: lightness 0.35-0.6, saturation at least 0.35. */
export function toAccent(c: Rgb): Rgb {
  const { h, s, l } = rgbToHsl(c)
  return hslToRgb(h, Math.max(s, 0.35), clamp(l, 0.35, 0.6))
}

const css = ({ r, g, b }: Rgb): string => `rgb(${r} ${g} ${b})`

export function accentVars(c: Rgb): { accent: string; gradTop: string } {
  const a = toAccent(c)
  const { h, s } = rgbToHsl(a)
  return { accent: css(a), gradTop: css(hslToRgb(h, s, 0.28)) }
}
```

```ts
export interface QueueState {

  index: number

  tracks: [...q.tracks, track]
```

```ts
  title?: string

  artists?: { name: string }[]

  if (!raw.id || !raw.title) return null
  const thumbs = raw.thumbnail?.contents ?? []
  const best = thumbs.reduce<{ url: string; width: number } | null>(

  )

  return {
    videoId: raw.id,
    title: raw.title,
    artist: artists.length ? artists.join(', ') : 'Unknown artist',
    coverUrl: best ? upsizeCover(best.url) : '',
    durationSec: raw.duration?.seconds ?? 0
  }
}
```

```ts
import http from 'node:http'
import type { AddressInfo } from 'node:net'
import { Readable } from 'node:stream'

const TTL_MS = 10 * 60 * 1000
const PASS_HEADERS = ['content-type', 'content-length', 'content-range', 'accept-ranges']

export function createStreamProxy(resolveUrl: (videoId: string) => Promise<string>): {
  start(): Promise<number>
  stop(): Promise<void>
} {
  const cache = new Map<string, { url: string; at: number }>()

  async function urlFor(id: string): Promise<string> {
    const hit = cache.get(id)
    if (hit && Date.now() - hit.at < TTL_MS) return hit.url
    const url = await resolveUrl(id)
    cache.set(id, { url, at: Date.now() })
    return url
  }

  const server = http.createServer(async (req, res) => {
    const m = /^\/stream\/([\w-]{11})$/.exec(req.url ?? '')
    if (!m) {
      res.writeHead(404).end()
      return
    }
    const id = m[1]
    const ac = new AbortController()
    res.on('close', () => ac.abort())
    try {
      const url = await urlFor(id)
      const upstream = await fetch(url, {
        headers: { range: req.headers.range ?? 'bytes=0-' },
        signal: ac.signal
      })
      if (upstream.status >= 400 || !upstream.body) {
        cache.delete(id)
        res.writeHead(502).end()
        return
      }
      const headers: Record<string, string> = { 'access-control-allow-origin': '*' }
      for (const h of PASS_HEADERS) {
        const v = upstream.headers.get(h)
        if (v) headers[h] = v
      }
      res.writeHead(upstream.status, headers)
      const body = Readable.fromWeb(upstream.body as never)
      body.on('error', () => res.destroy())
      body.pipe(res)
    } catch {
      cache.delete(id)
      if (!res.headersSent) res.writeHead(502)
      res.end()
    }
  })

  return {
    start: () =>
      new Promise((resolve) => {
        server.listen(0, '127.0.0.1', () => resolve((server.address() as AddressInfo).port))
      }),
    stop: () =>
      new Promise((resolve) => {
        server.closeAllConnections()
        server.close(() => resolve())
      })
  }
}
```

```ts
import type { Library } from '../library/library'
import type { PartnerPlaylist, PlaylistWithTracks, Track } from '../../shared/types'

export interface Remote {
  upsertPlaylist(p: PlaylistWithTracks): Promise<void>
  deletePlaylist(id: string): Promise<void>
  replaceLikes(tracks: Track[]): Promise<void>
  listPartnerPlaylists(): Promise<PartnerPlaylist[]>
}

export class SyncEngine {
  constructor(
    private lib: Library,
    private remote: Remote
  ) {}

  async flush(): Promise<{ pushed: number; error?: string }> {
    let pushed = 0
    for (const key of this.lib.pendingKeys()) {
      try {
        await this.push(key)
        this.lib.clearKey(key)
        pushed++
      } catch (e) {
        return { pushed, error: e instanceof Error ? e.message : String(e) }
      }
    }
    return { pushed }
  }

  private async push(key: string): Promise<void> {
    const i = key.indexOf(':')
    const kind = i === -1 ? key : key.slice(0, i)
    const id = i === -1 ? '' : key.slice(i + 1)
    if (kind === 'playlist') {
      const p = this.lib.getPlaylist(id)
      if (p) await this.remote.upsertPlaylist(p)
      else await this.remote.deletePlaylist(id)
    } else if (kind === 'likes') {
```

```ts
import { existsSync, mkdirSync, readFileSync, writeFileSync } from 'node:fs'
import { dirname } from 'node:path'

/** Minimal localStorage-like store so Supabase can keep the login session in Node. */
export class FileStorage {
  constructor(private path: string) {}

  private read(): Record<string, string> {
    try {
      return existsSync(this.path) ? JSON.parse(readFileSync(this.path, 'utf8')) : {}
    } catch {
      return {}
    }
  }

  private write(data: Record<string, string>): void {
    mkdirSync(dirname(this.path), { recursive: true })
    writeFileSync(this.path, JSON.stringify(data), { mode: 0o600 })
  }

  getItem(key: string): string | null {
    return this.read()[key] ?? null
  }

  setItem(key: string, value: string): void {
    this.write({ ...this.read(), [key]: value })
  }

  removeItem(key: string): void {
    const data = this.read()
    delete data[key]
    this.write(data)
  }
}
```

```ts
import { useEffect, useState } from 'react'
import type { Track } from '../../../shared/types'
import { accentVars, dominantColor } from './color'

export type ThemeName = 'dynamic' | 'dark' | 'light'
const STORAGE_KEY = 'phinas.theme'

function readStored(): ThemeName {
  try {
    const v = localStorage.getItem(STORAGE_KEY)
    if (v === 'dynamic' || v === 'dark' || v === 'light') return v
  } catch {
    /* storage unavailable */
  }
  return 'dynamic'
}

async function coverPixels(url: string): Promise<Uint8ClampedArray> {
  const img = new Image()
  img.crossOrigin = 'anonymous'
  img.src = url
  await img.decode()
  const canvas = document.createElement('canvas')
  canvas.width = 32
  canvas.height = 32
  const ctx = canvas.getContext('2d', { willReadFrequently: true })!
  ctx.drawImage(img, 0, 0, 32, 32)
  return ctx.getImageData(0, 0, 32, 32).data
}

/** Applies the theme to <html>. In Dynamic mode, accent colors follow the current cover. */
export function useTheme(track: Track | null): { theme: ThemeName; setTheme: (t: ThemeName) => void } {
  const [theme, setThemeState] = useState<ThemeName>(readStored)

  const setTheme = (t: ThemeName): void => {
    setThemeState(t)
    try {
      localStorage.setItem(STORAGE_KEY, t)
    } catch {
      /* ignore */
    }
  }

  useEffect(() => {
    const root = document.documentElement
    // Dynamic uses the dark base; light/dark are fixed.
    root.dataset.theme = theme === 'light' ? 'light' : 'dark'
    if (theme !== 'dynamic') {
      root.style.removeProperty('--accent')
      root.style.removeProperty('--grad-top')
      return
    }
    let cancelled = false
    const apply = (vars: { accent: string; gradTop: string }): void => {
      if (cancelled) return
      root.style.setProperty('--accent', vars.accent)
      root.style.setProperty('--grad-top', vars.gradTop)
    }
    if (!track?.coverUrl) {
      root.style.removeProperty('--accent')
      root.style.removeProperty('--grad-top')
      return
    }
    coverPixels(track.coverUrl)
      .then((px) => apply(accentVars(dominantColor(px))))
      .catch(() => {
        // CORS or decode failure: fall back to the default accent.
        if (cancelled) return
        root.style.removeProperty('--accent')
        root.style.removeProperty('--grad-top')
      })
    return () => {
      cancelled = true
    }
  }, [theme, track?.coverUrl])

  return { theme, setTheme }
}
```

```tsx
  track: Track | null
  playing: boolean
  time: number
  duration: number
  volume: number
  notice: string | null
  play(tracks: Track[], index: number): void
  addToQueue(track: Track): void
  toggle(): void
  next(): void
  prev(): void
  jump(index: number): void
  seek(sec: number): void
  setVolume(v: number): void
  dismissNotice(): void
}

const Ctx = createContext<PlayerApi | null>(null)

export function usePlayer(): PlayerApi {
  const v = useContext(Ctx)
  if (!v) throw new Error('usePlayer must be used inside PlayerProvider')
  return v
}

export function PlayerProvider({ children }: { children: ReactNode }): JSX.Element {
  const [audio] = useState(() => new Audio())
  const [queue, setQueue] = useState<QueueState>(empty)
  const [token, setToken] = useState(0)
  const [playing, setPlaying] = useState(false)
  const [time, setTime] = useState(0)
  const [duration, setDuration] = useState(0)
  const [volume, setVolumeState] = useState(0.8)
  const [notice, setNotice] = useState<string | null>(null)

  const track = current(queue)

  const advance = useCallback(() => {
    setQueue((q) => {
      const n = next(q)
      if (!n) {
        audio.pause()
        return q
      }
      return n
    })
  }, [audio])

  useEffect(() => {
    audio.volume = volume
  }, [audio, volume])

  useEffect(() => {
    const on = <K extends keyof HTMLMediaElementEventMap>(e: K, fn: () => void): (() => void) => {
      audio.addEventListener(e, fn)
      return () => audio.removeEventListener(e, fn)
    }
    const offs = [
      on('timeupdate', () => setTime(audio.currentTime)),
      on('durationchange', () => setDuration(Number.isFinite(audio.duration) ? audio.duration : 0)),
      on('play', () => setPlaying(true)),
      on('pause', () => setPlaying(false)),
      on('ended', advance)
    ]
    return () => offs.forEach((off) => off())
  }, [audio, advance])

  // Load and start whenever the current track (or a replay request) changes.
  useEffect(() => {
    if (!track) return
    let cancelled = false
    const skip = (): void => {
      if (cancelled) return
      setNotice(`Tidak bisa memutar "${track.title}", dilewati.`)
      advance()
    }
    window.phinas
      .streamUrl(track.videoId)
      .then((url) => {
        if (cancelled) return
        audio.src = url
        audio.onerror = skip
        return audio.play()
      })
      .catch(skip)
    return () => {
      cancelled = true
      audio.onerror = null
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [track?.videoId, queue.index, token])

  const api = useMemo<PlayerApi>(
    () => ({
      queue,
      track,
      playing,
      time,
      duration,
      volume,
      notice,
      play: (tracks, index) => {
        setQueue(startQueue(tracks, index))
        setToken((t) => t + 1)
      },
      addToQueue: (t) => setQueue((q) => (q.tracks.length ? enqueue(q, t) : startQueue([t], 0))),
      toggle: () => {
        if (!audio.src) return
        if (audio.paused) void audio.play()
        else audio.pause()
      },
      next: () => advance(),
      prev: () => {
        if (audio.currentTime > 3) audio.currentTime = 0
        else setQueue((q) => prev(q))
      },
      jump: (i) => setQueue((q) => jumpTo(q, i)),
      seek: (s) => {
        audio.currentTime = s
      },
      setVolume: setVolumeState,
      dismissNotice: () => setNotice(null)
    }),
    [queue, track, playing, time, duration, volume, notice, audio, advance]
  )

  return <Ctx.Provider value={api}>{children}</Ctx.Provider>
}
```

```tsx
export function TrackRow({ track, onPlay, onRemove, playlists }: Props): JSX.Element {
  const { track: playing } = usePlayer()
  const [liked, setLiked] = useState(false)

  useEffect(() => {
    let alive = true
    window.phinas.listLikes().then((l) => alive && setLiked(l.some((x) => x.videoId === track.videoId)))
    return () => {
      alive = false
    }
  }, [track.videoId])

  const toggleLike = async (): Promise<void> => {
    await window.phinas.setLiked(track, !liked)
    setLiked(!liked)
  }

  const addTo = async (playlistId: string): Promise<void> => {
    if (playlistId) await window.phinas.addTrack(playlistId, track)
  }

  return (
    <div className={`row${playing?.videoId === track.videoId ? ' playing' : ''}`}>
      <button className="cover" style={{ backgroundImage: `url("${track.coverUrl}")` }} onClick={onPlay} aria-label="Putar" />
      <div className="meta">
        <div className="title">{track.title}</div>
        <div className="muted">
          {track.artist} · {fmt(track.durationSec)}
        </div>
      </div>
      <div>
        <button className={`icon-btn${liked ? ' on' : ''}`} onClick={toggleLike} aria-label="Suka">
          {liked ? '♥' : '♡'}
        </button>
        {onRemove && (
          <button className="icon-btn" onClick={onRemove} aria-label="Hapus dari playlist">
            ✕
          </button>
        )}

              {p.name}
            </option>
          ))}
        </select>
      ) : (
        <span />
      )}
```

```tsx
      {status?.configured && !status.signedIn && (
        <form onSubmit={signIn} style={{ display: 'flex', flexDirection: 'column', gap: 8, maxWidth: 320 }}>
          <input type="email" placeholder="Email" value={email} onChange={(e) => setEmail(e.target.value)} required />
          <input type="password" placeholder="Password" value={password} onChange={(e) => setPassword(e.target.value)} required />
          {error && <div className="notice">{error}</div>}
          <button className="btn" type="submit">Masuk</button>
        </form>
      )}
    </>
  )
}
```
